	%On the numerical solution of differetial equations of Lane-Emden type
	%Problem 4.1
	\begin{equation}
		\begin{cases}\displaystyle
			\diff[2]{y\left(x\right)}{x}+
			\frac{2}{x}
			\diff{y\left(x\right)}{x}+
			y\left(x\right)=
			6 + 12 x+ x^{2} + x^{3}
			& \text { for }0 \leq x \leq 1. \\[0.5cm]
			\displaystyle
			y
			\left(0\right)=
			0,\quad
			\diff{y\left(0\right)}{x}=
			0.
			&
		\end{cases}
	\end{equation}
	\text{Where the exact solution is }
	$y\left(x\right) = x^{2} + x^{3}$.
	
	\newpage
	%Problem 4.2

			\frac{2}{x}
			\diff{y\left(x\right)}{x} =
			2 (2x^{2} + 3 )y\left(x\right)
			& \text { for }0 \leq x \leq 1. \\[0.5cm]
			\displaystyle
			y
			\left(0\right)=
			1,\quad
			\diff{y\left(0\right)}{x}=
			0.
			&
		\end{cases}
	\end{equation}
	\text{Where the exact solution is }
	$y\left(x\right) = e^{x^{2}}$.
