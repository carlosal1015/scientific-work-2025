\begin{frame}
    \frametitle{Problem statement}
    \begin{block}{Singular Cauchy problem of Lane-Emden type}
        Let be $k\geq 0$ a shape factor.
        \begin{equation}
            \begin{cases}\displaystyle
                \diff[2]{\theta\left(\xi\right)}{\xi}+
                \frac{k}{\xi}
                \diff{\theta\left(\xi\right)}{\xi}+
                f\left(\xi\right)
                g\left(\theta\right)=
                h\left(\xi\right)
                 & \text { for }0<\xi\leq R. \\\displaystyle
                \theta
                \left(0\right)=
                \theta_{0},\quad
                \diff{\theta\left(0\right)}{\xi}=
                \theta^{\prime}_{0}.
                 &
            \end{cases}.\label{eq:CauchyProblem}
        \end{equation}
    \end{block}

    \pause

    \begin{columns}[c]
        \begin{column}{0.63\textwidth}
            \begin{alertblock}{Common equations for $f\left(\xi\right)=1$ and $g\left(\theta\right)$~\cite{umesh}}
                \begin{description}
                    \item[$\theta^{m}$]

                          \eqref{eq:CauchyProblem} is called the first kind of Lane-Emden type equation.

                    \item[$\exp\left(\theta\right)$]

                          \eqref{eq:CauchyProblem} is called the second kind of Lane-Emden type equation.
                          Arises in modeling density diffusion in an isothermal gas sphere.
                          Analytical solution~\cite{alzate2014}.

                    \item[${\left(\theta^{2}-C\right)}^{\frac{3}{2}}$]

                          Is turned into the white-dwarf equation, which was introduced in %\cite{chandrasekhar1967} 
                          in his study of the gravitational potential of degenerate stars.
                \end{description}
            \end{alertblock}
