\section{Capitulo 1}
% ----------------------- Motivación Física -------------------

\begin{frame}{Motivación}

\begin{columns}[t] % [t] para alinear arriba
    \begin{column}{0.48\textwidth}
       En la teoría de estructuras estelares, la ecuación de Lane-Endem(L-E) ha sido utilizada para describir la variación de temperatura en una estrella con equilibrio hidrostatico bajo un modelo politrópico(revisar).
       %\cite{adomian1995} \cite{wazwaz}
    \end{column}

    \begin{column}{0.48\textwidth}
        La ecuación del L-E es esencialmente una consecuencia del equilibrio hidrostatico y de la Presión-Densidad 
        $$P=K\rho^{1+\frac{1}{n}}$$
        %\cite{maciel2007stellar}
        %\cite{alzate2014}
        \begin{itemize}
            \item $P(r)$: Presión.
            \item $\rho(r):$ Densidad.
            \item $n:$ Índice politrópico.
            \item $K:$ Constante
        \end{itemize}
    \end{column}
\end{columns}

\end{frame}





% -------------------- Ecuaciones base ------------------------
\begin{frame}{Ecuaciones fundamentales}
El modelo combina:

\begin{itemize}
    \begin{equation}
    \dv{P}{r} = -\frac{G M(r) \rho(r)}{r^2} \tag{Ecuación de equilibrio hidrostático}
    \end{equation}
    \begin{equation}
    \dv{M}{r} = 4\pi r^2 \rho(r) \tag{Ecuación de masa}
    \end{equation}
    \begin{equation}
     P = K \rho^{1+1/n} \tag{Ecuación de estado politrópica}
    \end{equation}
\end{itemize}

Estas ecuaciones llevan, tras adimensionalización, a la Ecuación de Lane--Emden.
\end{frame}	
% --------------------- Lane-Emden ----------------------------
\begin{frame}{Ecuación general tipo Lane--Emden}

\begin{block}{Ecuación diferencial}
\[
w''(r) + \frac{P}{r}\, w'(r) + N\!\left(w(r)\right) = g(r),
\qquad 0 < r \le R,0\le P
\tag{1}
\]
\end{block}

\begin{block}{Condiciones iniciales}
\[
w(0) = A, 
\qquad 
w'(0) = B.
\tag{2}
\]
\end{block}

\begin{block}{Interpretación de los términos}
\[
N = \text{nonlinear term},\qquad
P = \text{shape factor},\qquad
A,B = \text{constants}.
\]
\end{block}
\end{frame}

\begin{frame}{\(N(w)=w^{m}\)}
     Si \(N(w)=w^{m}\), entonces la ecuación (1) es la ecuación de Lane--Emden de 
    (1), la cual surge en el modelamiento de la teoría de la 
    estructura estelar.%\cite{umesh}
\end{frame}

\begin{frame}{\(N(w)=e^{w}\)}
    si \(N(w)=e^{w}\), entonces la ecuación (1) es la 
    ecuación de Lane--Emden de (1), que aparece en el 
    modelamiento de la difusión de densidad en una esfera de gas isotérmica. %\cite{umesh}

    Existe una demostración física en el paper de Alzate %\cite{alzate2014}
\end{frame}

\begin{frame}{\(N(w)=(y^2-C)^{\frac{3}{2}}\)}
     Now consider \( g(y) = (y^2-C)^{3/2} \), then Equation~(1) is turned into the white-dwarf equation, which was introduced in %\cite{chandrasekhar1967} 
     in his study of the gravitational potential of degenerate stars.
\end{frame}

% ------------------ Nuestra ecuación tipo Lane-Endem ---------------------
\begin{frame}{Nuestra ecuación tipo Lane-Endem}
La forma equivalente más usada es:
\[
    \theta''(\xi) + \frac{2}{\xi}\theta'(\xi) + \theta^n = 0.
\]

Con condiciones iniciales:
\[
    \theta(0) = 1, \qquad \theta'(0)=0.
\]

\begin{itemize}
    \item Garantizan regularidad en el origen.
    \item Evitan la singularidad del término \(2/\xi\).
\end{itemize}

\end{frame}

\begin{frame}{Métodos}
    La principal dificultad al resolver ecuaciones diferenciales singulares es el 
comportamiento singular que ocurre en \(r = 0\). Sin embargo, se han utilizado 
muchos métodos para resolver este tipo de ecuaciones diferenciales singulares. 
Por ejemplo, Wazwaz (2005) propuso un nuevo esquema; Khan \textit{et al}. (2012) 
aplicaron el método de transformación diferencial (DTM), y Chowdhury y Hashim 
(2009) utilizaron el método de perturbación por homotopía (HPM) para obtener la 
solución aproximada del problema (1). 

Baranwal \textit{et al}. (2012) aplicaron un enfoque híbrido, y Parand \textit{et al}. 
(2009) utilizaron una técnica seudoespectral basada en las funciones de Legendre 
racionales y la integración de Gauss--Radau para la solución de la ecuación de 
Lane--Emden de tipo (1). 

Recientemente, Saeed Mutaish y Hasan (2018), y Hasan y Saeed Mutaish (2018) 
resolvieron ecuaciones de tipo Lane--Emden mediante un método de descomposición 
de Adomian ajustado y también describieron la formación de estas ecuaciones.
%\cite{umesh}
\end{frame}

\begin{frame}{Soluciones conocidas}
En general no hay soluciones analíticas a la ecuación de Lane-Endem para un índice m arbitrario.
Pero para algunos casos específicos, es sencillo obtenerlos,por ejemplo: para n=0,1,5. 
%\cite{maciel2007stellar}
%Muestran las soluciones y además dan su interpretación.
\begin{itemize}
    \item \(n=0\): \(\displaystyle \theta = 1 - \frac{\xi^2}{6}\)
    \item \(n=1\): \(\displaystyle \theta = \frac{\sin\xi}{\xi}\)
    \item \(n=5\): \(\displaystyle \theta = \frac{1}{\sqrt{1 + \xi^2/3}}\)
\end{itemize}
