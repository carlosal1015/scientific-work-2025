    \headerbox{\bf\color{ctitle}Introduction}{name=introduction,column=0,row=0,span=20}{
        \textbf{Given a set of Stellar Structure Equations}\footnotesize
        \begin{equation*}
            \left.\begin{aligned}
                \diff{P}{r} & = -\frac{GM\rho}{r^{2}}.                \\
                \diff{M}{r} & = 4\pi r^{2}\rho.                       \\
                P           & = K{\left(\rho\right)}^{1+\frac{1}{n}}.
            \end{aligned}\right\}
            \!\!\!\implies\!\!\!
            \begin{aligned}
                \frac{1}{r^{2}}
                \diff*{
                    \left(
                    \frac{r^{2}}{\rho}\diff{P}{r}
                    \right)
                }{r}=
                -4\pi G \rho.
                % \text{See the model derivation.}
            \end{aligned}
        \end{equation*}\normalsize

        \

        \textbf{Goal}\small

        Obtain numerical solutions for Lane-Emden equations using efficient and accurate methods.\normalsize

        \

        \textbf{Existing methods}\small
        \begin{itemize}
            \item Transformation to 1st-order ODE system for obtain numerical solution with standard ODE solvers.

            \item Adomian Decomposition Method to aproximate the analytical series solution.
                  % \item Differential transform method $\implies$ Series-based, limited convergence for certain $n$.
        \end{itemize}\normalsize

        \textbf{Contributions}\small
        \begin{itemize}
            \item

                  Theoretical analysis of existence, uniqueness, and error bounds for solutions.

            \item Comparison across multiple polytropic indices $n$ between Adomian decomposition method
                  with Dormand-Prince RK5(4)7M.
        \end{itemize}
    }

    \headerbox{\bf\color{ctitle}Theoretical Analysis}{name=theorems,column=20,row=0,span=29}{
        \small
        \setlength{\abovedisplayskip}{3pt}
        \setlength{\belowdisplayskip}{3pt}
        \setlength{\abovedisplayshortskip}{1pt}
        \setlength{\belowdisplayshortskip}{1pt}

        \begin{theorem}[\colorbox{mathcolor}{\bfseries Existence and Uniqueness}]
            Let $m\in\mathbb{N}$.
            If $g\in C^{1}$,
            then $\forall\alpha\in\mathbb{R}$,
            $\exists! u\in C^{2}\left(\mathbb{R},\mathbb{R}\right)$ such that
            \begin{equation}
                \left\{
                \begin{aligned}
                    \diff*{\left(\xi^{m-1}\diff{u}{\xi}\right)}{\xi} & =
                    \xi^{m-1} g\left(u,\xi\right).                       \\
                    u\left(0\right)=\alpha,\quad \diff.|.{u}{\xi}[0] & =
                    0.
                \end{aligned}
                \right.\label{eq:1}
            \end{equation}
        \end{theorem}

        \begin{proof}
            Let $v\coloneqq\diff{u}{\xi}$.
            % The problem~\eqref{eq:1}
            % implies the integral equation
            % \vspace*{-.5\baselineskip}\setlength\belowdisplayshortskip{0pt}
            % \begin{equation*}
            %     \left\{
            %     \begin{aligned}
            %         \diff{u}{\xi}                        & =
            %         v.                                       \\
            %         \diff*{\left(\xi^{m-1}v\right)}{\xi} & =
            %         \xi^{m-1}
            %         g\left(u,\xi\right).                     \\
            %         u\left(0\right)=\alpha,
            %         v\left(0\right)                      & =
            %         0.
            %     \end{aligned}
            %     \right.\iff
            %     \left\{
            %     \begin{aligned}
            %         u\left(\xi\right) & =
            %         \alpha+
            %         \int_{0}^{\xi}
            %         v\left(t\right)\dl t. \\
            %         v\left(\xi\right) & =
            %         \frac{1}{\xi^{m-1}}
            %         \int_{0}^{\xi}
            %         t^{m-1}
            %         g\left(u,t\right)\dl t.
            %     \end{aligned}
            %     \right.
            % \end{equation*}
            Let $T>0$, $X_{T}\coloneqq C\left(\left[0,T\right]\right)\times C\left(\left[0,T\right]\right)$.\newline
            For $\left(u,v\right)\in X_{T}$:
            \begin{math}
                {\left\|\left(u,v\right)\right\|}_{T}\coloneqq
                \max
                \left\{
                {\left\|u\right\|}_{\infty},
                {\left\|v\right\|}_{\infty}
                \right\}
            \end{math}.\newline
            \textbf{Claim:} $\left(X_{T},{\left\|\cdot\right\|}_{T}\right)$ is a Banach space.
            Let be $B\subset X_{T}$ closed ball.
            Define
            \begin{align*}
                \Lambda\colon B & \longrightarrow B \\
                \left(u,v\right)
                                & \longmapsto
                \left(
                \alpha+
                \int_{0}^{\xi}
                v\left(t\right)\dl t,
                \frac{1}{\xi^{m-1}}
                \int_{0}^{\xi}
                t^{m-1}g\left(u,t\right)\dl t
                \right).
            \end{align*}
            $\Lambda$ is a contraction, $\left|\operatorname{Fix}\left(\Lambda\right)\right|=1$
            by the Banach fixed-point theorem.
        \end{proof}

        \begin{theorem}[\colorbox{mathcolor2}{\bfseries Error bound}]
            Let $u\in C^{n+1}\left(\left[0,1\right]\right)$ the exact
            solution of~\eqref{eq:1} and $u_{n}=\sum_{i=0}^{n}c_{i}t^{i}$
            their $n$-th order approximation.
            Then,
            \begin{equation*}
                \left\|u\left(t\right)-u_{n}\left(t\right)\right\|_{\infty}\leq
                \frac{M}{\left(n+1\right)!}+
                \max_{0\leq i\leq n}
                \left|a_{i}\right|,
            \end{equation*}
            where
            \begin{math}
                M=
                \max_{0\leq t\leq 1}
                \left|
                u^{\left(n+1\right)}\left(t\right)
                \right|
            \end{math}
            and
            \begin{math}
                a_{i}=
                \sum_{i=0}^{n}
                \left[
                    \frac{u^{\left(i\right)}\left(0\right)}{i!}-
                    c_{i}
                    \right]
            \end{math}.
        \end{theorem}
    }

    \headerbox{\bf\color{ctitle}Adomian Decomposition Method~\cite{Adomian1995,Wazwaz2001}}{name=adomian,column=49,row=0,span=27}{
    \small
    \setlength{\abovedisplayskip}{3pt}
    \setlength{\belowdisplayskip}{3pt}
    \setlength{\parskip}{2pt}

    The general operator equation is written as
    \begin{equation}\label{eq:generaloperator}
        Lu+Ru+Nu=g.
    \end{equation}
    Where the operators represent
    \vspace*{-\baselineskip}\setlength\belowdisplayshortskip{0pt}
    \begin{multicols}{2}
        \begin{itemize}
            \setlength\itemsep{0em}
            \item $L$: Highest order derivative.

            \item $R$: Lower order derivatives.

            \item $N$: Nonlinear terms.

            \item $g$: Source term.
        \end{itemize}
    \end{multicols}
    \vspace*{-\baselineskip}\setlength\belowdisplayshortskip{0pt}
    Applying $L^{-1}$ to~\eqref{eq:generaloperator}, we obtain
    \begin{equation*}
        u=
        \phi+
        L^{-1}g-
        L^{-1}Ru-
        L^{-1}Nu.
    \end{equation*}
    \begin{definition}[\bfseries Adomian polynomials $A_{n}$ and $L^{-1}$ for Lane-Emden]
        \begin{align*}
            \forall n\in\mathbb{N}\cup\left\{0\right\}:
            A_{n}  & \coloneqq
            \frac{1}{n!}
            \diff*[n]{
                \left[
                    N\left(
                    \sum_{i=0}^{\infty}
                    \lambda^{i}
                    u_{i}
                    \right)
            \right]_{\lambda=0}}{\lambda}. \\
            L^{-1} & \coloneqq
            \int_{0}^{\xi}
            \xi^{-2}
            \int_{0}^{\xi}
            \xi^{2}\left(\cdot\right)
            \dl \xi
            \dl \xi.
        \end{align*}
    \end{definition}
    \vspace*{-.5\baselineskip}\setlength\belowdisplayshortskip{0pt}
    Decomposing $u=\sum\limits_{n=0}^{\infty}u_{n}$,
    the recursive scheme is
    \begin{equation}\label{eq:adm_system}
        \boxed{
            \left\{
            \begin{aligned}
                \forall n\in\mathbb{N}\cup\left\{0\right\}:
                u_{n+1} & =
                -L^{-1}Ru_{n}-L^{-1}A_{n}. \\
                u_{0}   & =
                \phi+L^{-1}g.
            \end{aligned}
            \right.
        }
    \end{equation}
    }
