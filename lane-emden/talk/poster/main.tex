    % BOX 1: Adomian decomposition method
    \headerbox{\bf\color{ctitle}Adomian decomposition method}{name=summary,column=0,row=0,span=30}{

        \small 
        \setlength{\abovedisplayskip}{3pt}
        \setlength{\belowdisplayskip}{3pt}
        \setlength{\parskip}{2pt}

        The general operator equation is $Lu+Ru+Nu=g$. Applying $L^{-1}$:
        \begin{equation*}
            u = \phi + L^{-1}g - L^{-1}Ru - L^{-1}Nu
        \end{equation*}
        
        \vspace{-0.3cm} 
        \begin{multicols}{2}
            \begin{itemize}
                \setlength\itemsep{0em} 
                \item $L$: Highest order derivative.
                \item $R$: Linear remainder.
                \item $N$: Nonlinear operator.
                \item $g$: Source term.
            \end{itemize}
        \end{multicols}
        \vspace{-0.2cm}

        We define the Adomian Polynomials $A_n$ and the operator $L^{-1}$ for Lane-Emden as:
            \begin{align*}
                A_n &= \frac{1}{n!} \frac{d^n}{d\lambda^n} \left[ N\left(\sum_{i=0}^\infty \lambda^i u_i\right) \right]_{\lambda=0}, \\
                L^{-1}(\cdot) &= \int_{0}^{\xi} s^{-2} \left( \int_{0}^{s} t^{2}(\cdot) \dl t \right) \dl s.
            \end{align*}
        Decomposing $u = \sum_{n=0}^\infty u_n$, the recursive scheme is:
        \begin{equation}\label{eq:adm_system}
            \left\{
            \begin{aligned}
                u_0     &= \phi + L^{-1}g \\
                u_1     &= -L^{-1}Ru_0 - L^{-1}A_0 \\
                u_2     &= -L^{-1}Ru_1 - L^{-1}A_1 \\
                        &\vdots \\
                u_{n+1} &= -L^{-1}Ru_n - L^{-1}A_n 
            \end{aligned}
            \right.
        \end{equation}


    }

    % % BOX 4: EXISTENCE AND UNIQUENESS
    \headerbox{\bf\color{ctitle}Existence, Uniqueness and Error}{name=algorithm,column=0,row=0.43,span=30}{
        \small 
        \setlength{\abovedisplayskip}{3pt}
        \setlength{\belowdisplayskip}{3pt}
        \setlength{\abovedisplayshortskip}{1pt}
        \setlength{\belowdisplayshortskip}{1pt}

        \begin{theorem}[\textbf{\(\highlight[mathcolor]{\text{Existence and uniqueness}}\)}]
            Let $m\in\mathbb{N}$. If $g\in C^{1}\left(\mathbb{R}\times\left[0,\infty\right), \mathbb{R}\right)$, then $\forall d\in\mathbb{R}$, $\exists! u\in C^{2}\left(\mathbb{R},\mathbb{R}\right)$ such that:
            \vspace{-0.05cm} 
            \begin{equation}
                \left\{
                \begin{aligned}
                    \diff*{\left(\xi^{m-1}\diff{u}{\xi}\right)}{\xi} & = \xi^{m-1} g\left(u,\xi\right). \\
                    u\left(0\right)=d,\quad \diff.|.{u}{\xi}[0] & =0.
                \end{aligned}
                \right.\label{eq:1}
            \end{equation}
        \end{theorem}
        \vspace{-0.3cm} 

        \begin{proof}
            Let $v\coloneqq\diff{u}{\xi}$. The problem~\eqref{eq:1} implies:
            \begin{equation*} 
                \left\{
                \begin{aligned}
                    \diff{u}{\xi} &= v. \\
                    \diff*{\left(\xi^{m-1}v\right)}{\xi} &= \xi^{m-1} g\left(u,\xi\right). \\
                    u(0)=d, v(0) &=0.
                \end{aligned}
                \right.\implies
                \left\{
                \begin{aligned}
                    u(\xi) &= d+ \int_{0}^{\xi}v(t)\dl t. \\
                    v(\xi) &= \frac{1}{\xi^{m-1}} \int_{0}^{\xi} t^{m-1}g(u,t)\dl t.
                \end{aligned}
                \right.
            \end{equation*}
            Let $\varepsilon>0$, $X_{\varepsilon}= C([0,\varepsilon])\times C([0,\varepsilon])$.\newline
            For $(u,v)\in X_{\varepsilon}$:
            ${\|(u,v)\|}_{\varepsilon}= \max \left\{ \max_{0\leq t\leq\varepsilon}|u(t)|, \max_{0\leq t\leq\varepsilon}|v(t)| \right\}$.\newline
            \textbf{Claim:} $\left(X_{\varepsilon},{\left\|\cdot\right\|}_{\varepsilon}\right)$ is a Banach space.\newline
            Define $T\colon K \longrightarrow K$ over closed set $K\subset X$:
            \vspace{-0.05cm}
            \begin{align*}
                (u,v) \longmapsto \left( d+\int_{0}^{\xi}v(t)\dl t, \quad \frac{1}{\xi^{m-1}} \int_{0}^{\xi} t^{m-1}g(u,t)\dl t \right)
            \end{align*}
            \vspace{-0.4cm} 
        \end{proof}
        \vspace{-0.2cm} 

        \begin{theorem}[\textbf{\(\highlight[mathcolor2]{\text{Error bound:}}\)}]
            Let $X(t)\in \mathcal{C}^{n+1}[0,1]$, and $X_n=\sum_{i=0}^{n}c_it^i$ be the exact and $n$-th order approx. solution of problem~\eqref{eq:1}. An upper bound of the absolute error is:
            \vspace{-0.1cm}
            $$\|X(t)-X_n(t)\|_\infty \leq \frac{M}{(n+1)!}+ \max\limits_{0\leq i\leq n}{|a_i|}$$
            where $M=\max_{0\leq t \leq 1}{|X^{(n+1)}(t)|}$ and $a_i=\sum_{i=0}^n(\frac{X^{(i)}(0)}{i!}-c_i)$.
        \end{theorem}
    }
