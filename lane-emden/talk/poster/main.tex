    % \textbf{\color{blue}AMP: Adversarial Model Perturbation}\\
    \begin{definition}[Power series]
        \begin{equation*}
            Lu+Ru+Nu=g\iff
            u=A+Bt+L^{-1}g-L^{-1}Ru-L^{-1}Nu,
        \end{equation*}
    \end{definition}
    where
    \begin{itemize}
        \item
              \(L\) is the highest order differential operator,
        \item
              \(R\) is the remainder of linear operator,
        \item
              \(N\) is the nonlinear operator,
        \item
              \(g\) is the source term.
    \end{itemize}
    The solution $u$ is decomposed into
    $\sum_{n=0}^{\infty}u_{n}$ with $u_{0}=A+Bt+L^{-1}g$.
    Then,
    \begin{equation*}
        \sum_{n=0}^{\infty}
        u_{n}=
        u_{0}-
        L^{-1}R\left(\sum_{n=0}^{\infty}u_{n}\right)-
        L^{-1}\left(\sum_{n=0}^{\infty}A_{n}\right).
    \end{equation*}
    Where $A_{n}$ are special Adomian polynomials.
    We use:
    \begin{equation*}
        L=
        \xi^{-2}(\diff{}{r})\xi^2(\diff{}{r})\iff
        L^{-1}=\int_{0}^{\xi}\xi^{-2}
        \left(\int_{0}^{\xi}\xi^{2}(\cdot)\dl\xi\right)
        \dl\xi.
    \end{equation*}

    \begin{definition}[Transformation]
        \begin{equation*}
            Y\left(k\right)=
            \frac{1}{k!}
            \left.\diff[k]{y(x)}{x}\right|_{x=x_0}
            \iff
            y\left(x\right)=
            \sum_{k=0}^{\infty}
            Y(k){(x-x_0)}^{k}.
        \end{equation*}
    \end{definition}
    The following theorems can be deduced:
    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=u(x)\pm v(x),\text{ then }Y(k)=U(k)\pm V(k).
        \end{equation*}
    \end{theorem}

    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=c u(x),\text{ then }Y(k)=c U(k). \quad c \in \mathbb{R}.
        \end{equation*}
    \end{theorem}

    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=u(x)v(x),\text{ then }Y(k)=\sum_{i=0}^{k}U(i)V(k-i).
        \end{equation*}
    \end{theorem}

    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=\diff[n]{u(x)}{x},\text{ then }Y(k)=\frac{(k+n)!}{k!}U(k+n).
        \end{equation*}
    \end{theorem}

    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=\frac{u(x)}{v(x)},\text{ then }Y(k)=\frac{1}{V(0)}\left(U(k)-\sum_{i=0}^{k-1}V(k-i)Y(i)\right).
        \end{equation*}
    \end{theorem}

    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=x^{m},\text{ then }Y(k)=\delta(k-m)=
            \begin{cases}
                1, & k=m;     \\
                0, & k\neq m.
            \end{cases}
        \end{equation*}
    \end{theorem}

    \begin{theorem}
        \begin{equation*}
            \text{If }y(x)=x^n u(x),\text{ then }Y(k)=
            \begin{cases}
                U(k-n), & k\geq n; \\
                0,      & k<n.
            \end{cases}
        \end{equation*}
    \end{theorem}

    \headerbox{\bf\color{ctitle}Differential Transformation method}{name=results,column=2,row=0,span=2}{
        \begin{theorem}[Existence and uniqueness of Cauchy problem of Lane-Emden]
            Let $m\in\mathbb{N}$.
            If
            \begin{math}
                f\in
                C^{1}
                \left(
                \mathbb{R}\times\left[0,\infty\right),
                \mathbb{R}
                \right)
            \end{math},
            then $\forall d\in\mathbb{R}$:
            $\exists! u\in C^{2}\left(\mathbb{R},\mathbb{R}\right)$ such that
            \begin{equation}
                \left\{
                \begin{aligned}
                    \diff*{\left(\xi^{m-1}\diff{u}{\xi}\right)}{\xi} & =
                    \xi^{m-1}
                    f\left(u,\xi\right).                                   \\
                    u\left(0\right)=d,\diff.|.{u}{\xi}[0]            & =0.
                \end{aligned}
                \right.\label{eq:1}
            \end{equation}
        \end{theorem}

        \begin{proof}
            Let $v\coloneqq\diff{u}{\xi}$.
            The problem~\eqref{eq:1}
            \begin{math}
                \left\{
                \begin{aligned}
                    \diff{u}{\xi}                        & =
                    v.                                         \\
                    \diff*{\left(\xi^{m-1}v\right)}{\xi} & =
                    \xi^{m-1}
                    f\left(u,\xi\right).                       \\
                    u\left(0\right)=d,v\left(0\right)    & =0.
                \end{aligned}
                \right.\implies
                \left\{
                \begin{aligned}
                    u\left(\xi\right) & =
                    d+
                    \int_{0}^{\xi}v\left(t\right)\dl t. \\
                    v\left(\xi\right) & =
                    \frac{1}{\xi^{m-1}}
                    \int_{0}^{\xi}
                    t^{m-1}f\left(u,t\right)\dl t.
                \end{aligned}
                \right.
            \end{math}.
            Let $\varepsilon>0$,
            \begin{math}
                X_{\varepsilon}=
                C\left(\left[0,\varepsilon\right]\right)\times
                C\left(\left[0,\varepsilon\right]\right)
            \end{math}
            and $\forall\left(u,v\right)\in X_{\varepsilon}$:
            \begin{math}
                {\left\|\left(u,v\right)\right\|}_{\varepsilon}=
                \max
                \left\{
                \max\limits_{0\leq t\leq\varepsilon}\left|u\left(t\right)\right|,
                \max\limits_{0\leq t\leq\varepsilon}\left|v\left(t\right)\right|
                \right\}
            \end{math}.
            \textbf{Claim:} $\left(X_{\varepsilon},{\left\|\cdot\right\|}_{\varepsilon}\right)$ is a Banach space.

            Define the operator $T$ over closed set $K\subset X$ to itself and apply the Banach Contraction Principle
            \begin{align*}
                T\colon K        & \longrightarrow K \\
                \left(u,v\right) & \longmapsto
                \left(
                d+\int_{0}^{\xi}v\left(t\right)\dl t,
                \frac{1}{\xi^{m-1}}
                \int_{0}^{\xi}
                t^{m-1}f\left(u,t\right)\dl t
                \right)
            \end{align*}
        \end{proof}

    \headerbox{\bf\color{ctitle}Causal History Model (\textsf{CHM})}{name=algorithm,column=0,row=0.4,span=1}{
        \begin{itemize}
            \item Causal History Model \textcolor{imptext}{extends the state-space modeling paradigm to video processing.} CHM processes videos online and in streaming fashion.
            \item \textbf{\textsf{CHM} models the evolving state and compensates the history for motion} relative to the input and learns to re-weight information in the history.
            \item The \textbf{causal design in \textsc{Turtle} enables recurrence in inference} through state-memorized historical features (stateful configuration); \textbf{while allowing parallel training} by sampling short video clips (stateless configuration).
            \item \textbf{\textsc{TURTLE}'s encoder processes each frame individually}, while its decoder reuses features from previously restored frames, based on the proposed Causal History Model (CHM).
        \end{itemize}
        \begin{align}
            \label{eq:chm}
            \vec{\hat{H}}_{t}^{[l]} & = \highlight[mathcolor]{\phi_{t}(\vec{H}_{t}^{[l]}, \vec{F}_{t}^{[l]})} \oplus \mathcal{B}_{t}(\vec{F}_{t}^{[l]}),   \\
            \label{eq:chm2}
            \vec{y}_t^{[l]}         & = \highlight[mathcolor2]{\psi_{t}(\vec{\hat{H}}_{t}^{[l]}, \vec{F}_{t}^{[l]})} + \mathcal{D}_{t}(\vec{F}_{t}^{[l]}).
        \end{align}
        \begin{itemize}
            \item \textbf{\(\highlight[mathcolor]{\text{State Align Block}~\phi}\)}: implicitly tracks and aligns the \textit{topk} most relevant regions in the history with the input frame. Therefore, \(\phi\) compensates for motion the input with respect to the trajectory and establishes \textit{one-to-topk} correspondence.
            \item \textbf{\(\highlight[mathcolor2]{\text{Frame History Router }~\psi}\)}: re-weights the motion compensated history along with the input, thereby encouraging \& accentuating relevant information.
        \end{itemize}
    }

    \headerbox{\bf\color{ctitle}Is \textsc{CHM} Necessary?}{name=method,column=1,row=0.4,span=1,boxpadding=0.45em}{
        \textbf{\color{blue}\textsf{CHM} Procedure:}
        % \textsf{CHM} utilizes attention mechanism to compute region-wise correspondences in the history, and to re-weight the entire trajectory. However, in the first case, CHM limits the lookback window to \textit{topk} most similar regions to avoid spurious correlations hampering the flow of useful information.
        \begin{itemize}
            \item \textsf{CHM} utilizes an attention mechanism to compute region-wise correspondences in the history and to re-weight the entire trajectory.
            \item However, \textsf{CHM} limits the lookback window to the \textit{top-k} most similar regions to avoid spurious correlations that may hamper the flow of useful information.
        \end{itemize}
